# Cafe24/GnuBoard data directory guard
#
# Copy this file to <gnuboard-root>/data/.htaccess after GnuBoard creates
# the data directory. The release zip intentionally excludes /data, so this
# guard must be installed during staging and again during production cutover.

Options -Indexes

<IfModule mod_mime.c>
    RemoveHandler .php .phtml .php3 .php4 .php5 .php7 .php8 .phar .cgi .pl
    RemoveType .php .phtml .php3 .php4 .php5 .php7 .php8 .phar .cgi .pl
</IfModule>

<IfModule mod_php.c>
    php_flag engine off
</IfModule>
<IfModule mod_php7.c>
    php_flag engine off
</IfModule>
<IfModule mod_php8.c>
    php_flag engine off
</IfModule>

<FilesMatch "\.(php|phtml|php[0-9]?|phar|cgi|pl|asp|aspx|jsp|jspx|html?|shtml|js|mjs|svg|svgz)$">
    <IfModule mod_authz_core.c>
        Require all denied
    </IfModule>
    <IfModule !mod_authz_core.c>
        Deny from all
    </IfModule>
</FilesMatch>
